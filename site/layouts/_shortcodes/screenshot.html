{{- /* A screen of the running program, in light and dark: {{< screenshot name="waiting" alt="What the screen shows" >}} inserts assets/screenshots/waiting-light.png and waiting-dark.png. "make screenshots" makes these files. They are not in git. The browser picks the image with the setting of the computer, as the site does with its colours. The width and the height keep the place of the image while it loads, so the text does not jump. */ -}} {{- $name := .Get "name" -}} {{- $alt := .Get "alt" -}} {{- if not $name -}} {{- errorf "%s: the shortcode screenshot needs a name" .Position -}} {{- else if not $alt -}} {{- errorf "%s: the screenshot %q needs alt text" .Position $name -}} {{- else -}} {{- $light := resources.Get (printf "screenshots/%s-light.png" $name) -}} {{- $dark := resources.Get (printf "screenshots/%s-dark.png" $name) -}} {{- if and $light $dark -}} {{- else -}} {{- errorf "%s: the screenshot %q has no images. Add the shot to TestScreenshots in internal/e2e, then run make -B screenshots. Plain make screenshots can do nothing, because the stamp file says that the images are current" .Position $name -}} {{- end -}} {{- end -}}
