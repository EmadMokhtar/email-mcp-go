{{- /* The links to the previous and the next page, in the order of page-order.html. No page writes them by hand. Call it with the page that is shown: partial "pager.html" . */ -}} {{- $shown := . -}} {{- $order := partial "page-order.html" . -}} {{- $prev := false -}} {{- $next := false -}} {{- range $i, $page := $order -}} {{- if $page.Eq $shown -}} {{- if gt $i 0 -}} {{- $prev = index $order (sub $i 1) -}} {{- end -}} {{- if lt (add $i 1) (len $order) -}} {{- $next = index $order (add $i 1) -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next }}
{{- with $prev }} Previous {{ .LinkTitle }} {{- end }} {{- with $next }} Next {{ .LinkTitle }} {{- end }}
{{- end }}
